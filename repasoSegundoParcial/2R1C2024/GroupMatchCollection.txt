public interface GroupMatchCollection<T> extends List<T> {

  public void addMatchPredicate(Predicate <T> tester);

  public Set<T> groupingByMatches();

}

public class GroupMatchCollectionImpl<T> extends ArrayList<T> implements GroupMatchCollection<T> {

  private Set<Predicate <T>> testers = new HashSet<>();

  public void addMatchPredicate(Predicate <T> tester) { testers.add(tester); }

  private testsPassed(T element) {
    int counter = 0;
    for (Predicate<T> test : testers) {
      counter += test.test(element) ? 1:0;
    }
    return counter;
  }

  public List <T> groupingByMatches() {
    // Estoy teniendo en cuenta que puede pasar que haya muchos espacios
    // vacios, por eso un TreeSet<>
    Set<Integer, List<T>> grouped = new TreeSet<>();
    for (T element : this) {
      grouped.putIfAbsent(testsPassed(element), new ArrayList<>());
      grouped.get(correspondingGroup).add(element);
    }
    return grouped;
  }

}
