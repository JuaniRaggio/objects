public class RouteCentral {

  // Calcula el costo de el vuelo de ida en base a las millas de ida y vuelta
  private UnaryOperation<Double> routeCost;
  private Map<String, Map<String, Integer>> flights;

  public RouteCentral(UnaryOperation<Double> oneWayCost) {
    flights = new HashMap<>();
    routeCost = oneWayCost;
  }

  private boolean isValidOrigin(String origin) {
    return flights.containsKey(origin);
  }

  private boolean isValidDestination(String destination) {
    return flights.get(origin).containsKey(destination);
  }

  private boolean isValidRoute(String origin, String destination) {
    return isValidOrigin(origin) && isValidDestination(destination);
  }

  public void addOneWayRoute(String origin, String destination, int miles) {
    flights.putIfAbsent(origin, new HashMap<>());
    flights.get(origin).putIfAbsent(destination, miles);
  }

  public void addTwoWayRoute(String origin, String destination, int miles) {
    addOneWayRoute(origin, destination, routeCost.apply(miles));
    addOneWayRoute(destination, origin, routeCost.apply(miles));
  }

  public void changeRouteCost(String origin, String destination, int miles) {
    if (isValidRoute(origin, destination)) {
      flights.get(origin).put(destination, miles);
      return;
    }
    throw new NoSuchElementException("Route does not exist");
  }

  public Integer getDirectRouteMiles(String origin, String destination) {
    if (!isValidRoute(origin, destination))
      throw new NoSuchElementException("Route does not exist");
    return flights.get(origin).get(destination);
  }

  public Integer getOneStopRouteMiles(String origin, String stop, String destination) {
    return getDirectRouteMiles(origin, stop) + getDirectRouteMiles(stop, destination);
  }

}
