public interface MatchCollection <T extends Comparable<? super T>> extends List <T> {

  protected Set <T> getMatchedWithOrder(Comparator <T> cmp);

  void addMathedPredicate(Predicate <T> pred);

  default Set <T> getAllMatched() {
    return getMatchedWithOrder(Comparator.naturalOrder());
  }

  default Set <T> getAllMatchedReversed() {
    return getMatchedWithOrder(Comparator.reverseOrder());
  }

}

public class MatchCollectionImpl <T> extends ArrayList <T> implements MatchCollection <T> {

  private Set<Predicate<T>> predicates = new HashSet<>();

  public addMatchedPredicate(Predicate <T> pred) {
    predicates.add(pred);
  }

  private boolean passesTest(T element) {
    for (Predicate <T> pred : predicates) {
      if (!pred.test(element))
        return false;
    }
    return true;
  }

  @Override
  protected Set<T> getMatchedWithOrder(Comparator <T> cmp) {
    TreeSet <T> orderedCollection = new TreeSet<>(cmp);
    for (T element : this) {
      if (passesTest(element))
        orderedCollection.add(element);
    }
  }

}
